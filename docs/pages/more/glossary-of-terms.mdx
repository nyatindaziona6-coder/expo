import React, { useState } from 'react';
import {
  SafeAreaView,
  View,
  Text,
  TextInput,
  ScrollView,
  TouchableOpacity,
  KeyboardAvoidingView,
  Platform,
  StatusBar,
  StyleSheet,
  Alert,
  Keyboard,
} from 'react-native';

const GREEN = '#087443';
const DARK = '#064D30';
const GOLD = '#F4C542';
const BG = '#F3F8F4';

const VEHICLES = [
  { name: 'Motorbike', icon: '🏍️', base: 10000, rate: 1200 },
  { name: 'Economy', icon: '🚗', base: 15000, rate: 1800 },
  { name: 'Comfort', icon: '🚘', base: 20000, rate: 2400 },
  { name: 'XL', icon: '🚙', base: 25000, rate: 3000 },
];

const money = (value) =>
  'MWK ' + Math.round(value).toLocaleString('en-US');

function InputField({ label, value, onChangeText, placeholder, keyboardType }) {
  return (
    <View style={styles.field}>
      <Text style={styles.label}>{label}</Text>
      <TextInput
        style={styles.input}
        value={value}
        onChangeText={onChangeText}
        placeholder={placeholder}
        placeholderTextColor="#89978D"
        keyboardType={keyboardType || 'default'}
        autoCorrect={false}
        returnKeyType="done"
      />
    </View>
  );
}

function Button({ title, onPress, light }) {
  return (
    <TouchableOpacity
      onPress={onPress}
      activeOpacity={0.8}
      style={[styles.button, light && styles.lightButton]}
    >
      <Text style={[styles.buttonText, light && styles.lightButtonText]}>
        {title}
      </Text>
    </TouchableOpacity>
  );
}

export default function App() {
  const [page, setPage] = useState('Home');

  const [pickup, setPickup] = useState('');
  const [destination, setDestination] = useState('');
  const [distance, setDistance] = useState('');
  const [vehicleName, setVehicleName] = useState('Economy');
  const [fareOffer, setFareOffer] = useState('');
  const [rides, setRides] = useState([]);

  const [name, setName] = useState('');
  const [phone, setPhone] = useState('');
  const [email, setEmail] = useState('');
  const [identity, setIdentity] = useState('');

  const [driverName, setDriverName] = useState('');
  const [driverPhone, setDriverPhone] = useState('');
  const [driverLicence, setDriverLicence] = useState('');
  const [licenceExpiry, setLicenceExpiry] = useState('');
  const [carMake, setCarMake] = useState('');
  const [carModel, setCarModel] = useState('');
  const [carYear, setCarYear] = useState('');
  const [carColour, setCarColour] = useState('');
  const [carRegistration, setCarRegistration] = useState('');
  const [ownership, setOwnership] = useState('I own this vehicle');

  const vehicle =
    VEHICLES.find((item) => item.name === vehicleName) || VEHICLES[1];

  const km = Number(distance);
  const validDistance = distance.trim() !== '' && Number.isFinite(km) && km > 0;
  const estimate = validDistance ? vehicle.base + vehicle.rate * km : null;

  function calculateFare() {
    Keyboard.dismiss();

    if (!pickup.trim() || !destination.trim()) {
      Alert.alert('Journey details needed', 'Enter your pickup and destination.');
      return;
    }

    if (!validDistance) {
      Alert.alert(
        'Enter distance',
        'Enter the approximate journey distance in kilometres. Automatic map routing will be added when a route service is connected.'
      );
      return;
    }

    setFareOffer(String(Math.round(estimate / 500) * 500));
  }

  function requestRide() {
    Keyboard.dismiss();

    if (!name.trim() || !phone.trim()) {
      Alert.alert('Passenger details needed', 'Open Account and enter your name and phone.');
      setPage('Account');
      return;
    }

    if (!validDistance || !pickup.trim() || !destination.trim()) {
      Alert.alert('Journey details needed', 'Complete your journey details first.');
      setPage('Home');
      return;
    }

    const offered = Number(fareOffer);

    if (!Number.isFinite(offered) || offered < estimate) {
      Alert.alert('Check your offer', 'Your offer must be at least ' + money(estimate) + '.');
      return;
    }

    setRides((previous) => [
      {
        id: String(Date.now()),
        pickup,
        destination,
        vehicle: vehicleName,
        distance: km,
        fare: offered,
      },
      ...previous,
    ]);

    Alert.alert(
      'Ride request saved',
      'This prototype has saved your request on this screen. It has not contacted a real driver.'
    );
    setPage('Rides');
  }

  function saveAccount() {
    if (!name.trim() || !phone.trim()) {
      Alert.alert('Missing details', 'Enter your name and phone number.');
      return;
    }

    Alert.alert(
      'Details entered',
      'This prototype does not yet send your details to a secure server or verify your account.'
    );
  }

  function applyDriver() {
    const required = [
      driverName,
      driverPhone,
      driverLicence,
      licenceExpiry,
      carMake,
      carModel,
      carYear,
      carColour,
      carRegistration,
    ];

    if (required.some((value) => !value.trim())) {
      Alert.alert('Complete the form', 'Fill in all driver and vehicle fields.');
      return;
    }

    Alert.alert(
      'Application prepared',
      'Driver verification and document upload are not connected yet. Your details have not been submitted to a company.'
    );
  }

  function Header() {
    return (
      <View style={styles.header}>
        <View style={styles.brandRow}>
          <View style={styles.logo}>
            <Text style={styles.logoText}>MR</Text>
          </View>
          <View style={{ flex: 1 }}>
            <Text style={styles.brand}>MALAWI RIDE</Text>
            <Text style={styles.tagline}>Your journey, your Malawi</Text>
          </View>
          <View style={styles.flag}>
            <View style={{ flex: 1, backgroundColor: '#111111' }} />
            <View style={{ flex: 1, backgroundColor: '#D71920' }} />
            <View style={{ flex: 1, backgroundColor: '#168A45' }} />
          </View>
        </View>
      </View>
    );
  }

  function HomePage() {
    return (
      <>
        <View style={styles.hero}>
          <Text style={styles.eyebrow}>WELCOME TO MALAWI RIDE</Text>
          <Text style={styles.heroTitle}>Where are you going?</Text>
          <Text style={styles.heroText}>
            Choose your ride, enter your journey and see your estimated fare.
          </Text>
          <Text style={styles.countryBadge}>🇲🇼 MADE FOR MALAWI</Text>
        </View>

        <View style={styles.card}>
          <Text style={styles.sectionTitle}>Plan your journey</Text>

          <InputField
            label="PICKUP LOCATION"
            value={pickup}
            onChangeText={setPickup}
            placeholder="Where should the driver meet you?"
          />

          <Text style={styles.hint}>
            GPS pickup is not connected in this version. Type your current area.
          </Text>

          <InputField
            label="DESTINATION"
            value={destination}
            onChangeText={setDestination}
            placeholder="Town, area or landmark in Malawi"
          />

          <InputField
            label="DISTANCE IN KILOMETRES"
            value={distance}
            onChangeText={setDistance}
            placeholder="Enter approximate distance, e.g. 12"
            keyboardType="decimal-pad"
          />

          <Text style={styles.label}>CHOOSE YOUR RIDE</Text>

          {VEHICLES.map((item) => {
            const selected = item.name === vehicleName;
            const fare = validDistance
              ? item.base + item.rate * km
              : null;

            return (
              <TouchableOpacity
                key={item.name}
                onPress={() => {
                  setVehicleName(item.name);
                  if (validDistance) {
                    setFareOffer(
                      String(
                        Math.round((item.base + item.rate * km) / 500) * 500
                      )
                    );
                  }
                }}
                style={[
                  styles.vehicle,
                  selected && styles.vehicleSelected,
                ]}
              >
                <Text style={styles.vehicleIcon}>{item.icon}</Text>
                <View style={{ flex: 1 }}>
                  <Text style={styles.vehicleName}>{item.name}</Text>
                  <Text style={styles.vehicleDetail}>
                    Base {money(item.base)} + {money(item.rate)}/km
                  </Text>
                </View>
                <View style={{ alignItems: 'flex-end' }}>
                  {fare !== null && (
                    <Text style={styles.vehicleFare}>{money(fare)}</Text>
                  )}
                  <Text style={selected ? styles.selectedMark : styles.unselectedMark}>
                    {selected ? '●' : '○'}
                  </Text>
                </View>
              </TouchableOpacity>
            );
          })}

          <Button title="Calculate my fare" onPress={calculateFare} />

          {estimate !== null && fareOffer !== '' && (
            <View style={styles.fareBox}>
              <Text style={styles.fareLabel}>ESTIMATED FARE</Text>
              <Text style={styles.fareValue}>{money(estimate)}</Text>
              <Text style={styles.hint}>
                Estimated distance: {km.toFixed(1)} km. Rates are sample starting
                estimates and must be reviewed before launch.
              </Text>

              <InputField
                label="YOUR FARE OFFER (MWK)"
                value={fareOffer}
                onChangeText={setFareOffer}
                placeholder="Enter your offer"
                keyboardType="numeric"
              />

              <Button title="Request this ride" onPress={requestRide} />
            </View>
          )}
        </View>

        <View style={styles.infoCard}>
          <Text style={styles.infoTitle}>Safe journeys start here</Text>
          <Text style={styles.infoText}>
            A real launch still needs verified drivers, live matching, secure
            accounts, emergency support and payment services.
          </Text>
        </View>
      </>
    );
  }

  function RidesPage() {
    return (
      <View style={styles.card}>
        <Text style={styles.pageTitle}>My rides</Text>
        <Text style={styles.pageSubtitle}>
          Your requests saved during this app session.
        </Text>

        {rides.length === 0 ? (
          <View style={styles.empty}>
            <Text style={{ fontSize: 38 }}>🚗</Text>
            <Text style={styles.vehicleName}>No rides yet</Text>
            <Text style={styles.hint}>Plan a journey to get started.</Text>
            <Button title="Plan a ride" onPress={() => setPage('Home')} />
          </View>
        ) : (
          rides.map((ride) => (
            <View key={ride.id} style={styles.rideCard}>
              <Text style={styles.vehicleName}>{ride.vehicle}</Text>
              <Text style={styles.rideLine}>Pickup: {ride.pickup}</Text>
              <Text style={styles.rideLine}>Destination: {ride.destination}</Text>
              <Text style={styles.rideLine}>Distance: {ride.distance} km</Text>
              <Text style={styles.rideFare}>{money(ride.fare)}</Text>
              <Text style={styles.pending}>
                Saved locally — driver not contacted
              </Text>
            </View>
          ))
        )}
      </View>
    );
  }

  function AccountPage() {
    return (
      <View style={styles.card}>
        <Text style={styles.pageTitle}>My account</Text>
        <Text style={styles.pageSubtitle}>
          Enter your passenger details.
        </Text>

        <InputField label="FULL NAME" value={name} onChangeText={setName} placeholder="Your full name" />
        <InputField label="PHONE NUMBER" value={phone} onChangeText={setPhone} placeholder="+265..." keyboardType="phone-pad" />
        <InputField label="EMAIL ADDRESS" value={email} onChangeText={setEmail} placeholder="you@example.com" keyboardType="email-address" />
        <InputField label="NATIONAL ID / PASSPORT" value={identity} onChangeText={setIdentity} placeholder="Document number" />

        <View style={styles.notice}>
          <Text style={styles.noticeTitle}>Prototype notice</Text>
          <Text style={styles.noticeText}>
            These details remain in app memory only. SMS, email and identity
            verification are not connected. Do not enter sensitive information
            into this prototype.
          </Text>
        </View>

        <Button title="Save details for this session" onPress={saveAccount} />
        <Button
          title="Become a driver"
          light
          onPress={() => setPage('Driver')}
        />
      </View>
    );
  }

  function DriverPage() {
    return (
      <View style={styles.card}>
        <Text style={styles.pageTitle}>Become a driver</Text>
        <Text style={styles.pageSubtitle}>
          Complete the form to prepare your application.
        </Text>

        <InputField label="FULL NAME" value={driverName} onChangeText={setDriverName} placeholder="Name on your ID" />
        <InputField label="PHONE NUMBER" value={driverPhone} onChangeText={setDriverPhone} placeholder="+265..." keyboardType="phone-pad" />
        <InputField label="DRIVING LICENCE NUMBER" value={driverLicence} onChangeText={setDriverLicence} placeholder="Licence number" />
        <InputField label="LICENCE EXPIRY DATE" value={licenceExpiry} onChangeText={setLicenceExpiry} placeholder="DD/MM/YYYY" />
        <InputField label="VEHICLE MAKE" value={carMake} onChangeText={setCarMake} placeholder="e.g. Toyota" />
        <InputField label="VEHICLE MODEL" value={carModel} onChangeText={setCarModel} placeholder="e.g. Corolla" />
        <InputField label="VEHICLE YEAR" value={carYear} onChangeText={setCarYear} placeholder="e.g. 2020" keyboardType="numeric" />
        <InputField label="VEHICLE COLOUR" value={carColour} onChangeText={setCarColour} placeholder="e.g. White" />
        <InputField label="REGISTRATION NUMBER" value={carRegistration} onChangeText={setCarRegistration} placeholder="Vehicle plate number" />

        <Text style={styles.label}>VEHICLE PERMISSION</Text>
        {['I own this vehicle', 'I have the owner’s permission'].map((option) => (
          <TouchableOpacity
            key={option}
            style={styles.choice}
            onPress={() => setOwnership(option)}
          >
            <Text style={styles.choiceText}>
              {ownership === option ? '🔘' : '⚪'}  {option}
            </Text>
          </TouchableOpacity>
        ))}

        <View style={styles.notice}>
          <Text style={styles.noticeTitle}>Driver verification required</Text>
          <Text style={styles.noticeText}>
            This form does not upload documents or verify identity, licence,
            vehicle registration or ownership. Secure checks are required before
            accepting real passengers.
          </Text>
        </View>

        <Button title="Prepare application" onPress={applyDriver} />
      </View>
    );
  }

  const tabs = [
    { name: 'Home', icon: '⌂', label: 'Home' },
    { name: 'Rides', icon: '↗', label: 'My Rides' },
    { name: 'Driver', icon: '🚘', label: 'Drive' },
    { name: 'Account', icon: '○', label: 'Account' },
  ];

  return (
    <SafeAreaView style={styles.safe}>
      <StatusBar barStyle="light-content" backgroundColor={DARK} />

      <KeyboardAvoidingView
        style={{ flex: 1 }}
        behavior={Platform.OS === 'ios' ? 'padding' : undefined}
        keyboardVerticalOffset={0}
      >
        <Header />

        <ScrollView
          style={{ flex: 1 }}
          contentContainerStyle={styles.content}
          keyboardShouldPersistTaps="handled"
          keyboardDismissMode="on-drag"
          showsVerticalScrollIndicator={false}
        >
          {page === 'Home' && <HomePage />}
          {page === 'Rides' && <RidesPage />}
          {page === 'Account' && <AccountPage />}
          {page === 'Driver' && <DriverPage />}

          <Text style={styles.footer}>MALAWI RIDE • PROTOTYPE</Text>
        </ScrollView>

        <View style={styles.nav}>
          {tabs.map((tab) => {
            const active = page === tab.name;
            return (
              <TouchableOpacity
                key={tab.name}
                style={styles.navItem}
                onPress={() => {
                  Keyboard.dismiss();
                  setPage(tab.name);
                }}
              >
                <Text style={[styles.navIcon, active && styles.navActive]}>
                  {tab.icon}
                </Text>
                <Text style={[styles.navLabel, active && styles.navActive]}>
                  {tab.label}
                </Text>
                {active && <View style={styles.indicator} />}
              </TouchableOpacity>
            );
          })}
        </View>
      </KeyboardAvoidingView>
    </SafeAreaView>
  );
}
